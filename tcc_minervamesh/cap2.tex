\section{Introdução}

\paragraph{}A modelagem matemática de fenômenos físicos em engenharia frequentemente recai sobre a solução de Equações Diferenciais Parciais (EDPs). Neste trabalho, dois métodos numéricos clássicos são explorados: o Método das Diferenças Finitas (MDF) e o Método dos Elementos Finitos (MEF). Embora o MDF seja utilizado para problemas introdutórios 1D devido à sua simplicidade conceitual, o foco central desta plataforma — e sua robustez para geometrias complexas — reside no Método dos Elementos Finitos. Este capítulo detalha a formulação matemática e a discretização empregadas no \textit{MinervaMesh}.

\section{Equações Governantes}

\subsection{Transferência de Calor}
\paragraph{}A condução de calor em um domínio $\Omega$ é regida pela equação da difusão:
\begin{equation}
    \rho c_p \frac{\partial T}{\partial t} = \nabla \cdot (k \nabla T) + \dot{q}
    \label{eq:calor}
\end{equation}
\noindent onde $T$ é a temperatura, e $\dot{q}$ a fonte de calor.

\subsection{Mecânica dos Fluidos (Formulação $\psi-\omega$)}
\paragraph{}Para escoamentos 2D incompressíveis, utilizamos a formulação Vorticiade-Corrente, que desacopla a pressão e garante a conservação de massa automaticamente. As equações governantes são a Poisson para a função corrente ($\psi$) e o transporte para a vorticidade ($\omega$):

\begin{equation}
    \nabla^2 \psi = -\omega
    \label{eq:poisson_psi}
\end{equation}

\begin{equation}
    \frac{\partial \omega}{\partial t} + \mathbf{u} \cdot \nabla \omega = \frac{1}{Re} \nabla^2 \omega
    \label{eq:transp_omega}
\end{equation}

\section{Método dos Elementos Finitos (MEF)}

\paragraph{}O MEF é escolhido pela sua versatilidade em tratar contornos irregulares através de malhas não-estruturadas. A implementação segue a formulação de Galerkin padrão para problemas difusivos e estabilização para termos convectivos.

\subsection{Discretização Espacial: Elementos Triangulares Lineares (P1)}
\paragraph{}O domínio $\Omega$ é subdividido em uma malha de $N_{el}$ elementos triangulares. Em cada elemento $e$, a variável de campo $\phi$ (seja $T$, $\psi$ ou $\omega$) é aproximada linearmente:
\begin{equation}
    \phi^e(x, y) = \sum_{i=1}^{3} N_i^e(x, y) \phi_i^e
\end{equation}
\noindent onde $\phi_i^e$ são os valores nodais e $N_i^e$ são as funções de forma. Para um triângulo linear, as funções de forma são coordenadas de área, variando linearmente de 1 no nó $i$ até 0 na aresta oposta \cite{Zienkiewicz2013}.

\subsection{Formulação Fraca e Matrizes Elementares}
\paragraph{}Aplicando o Método dos Resíduos Ponderados de Galerkin à equação de Poisson geral ($-\nabla^2 \phi = f$), multiplicamos a equação residual por uma função teste $W = N_i$ e integramos sobre o domínio. Utilizamos o Teorema de Green (integração por partes) para reduzir a ordem da derivada no termo difusivo:

\begin{equation}
    \int_{\Omega} \nabla N_i \cdot \nabla \phi \, d\Omega = \int_{\Omega} N_i f \, d\Omega + \oint_{\Gamma} N_i \frac{\partial \phi}{\partial n} \, d\Gamma
\end{equation}

\paragraph{}Substituindo a aproximação discreta $\phi = \sum N_j \phi_j$, obtemos o sistema linear $\mathbf{K}\mathbf{\Phi} = \mathbf{F}$, onde a matriz de rigidez elementar $\mathbf{K}^e$ é:

\begin{equation}
    K_{ij}^e = \int_{\Omega_e} \left( \frac{\partial N_i}{\partial x}\frac{\partial N_j}{\partial x} + \frac{\partial N_i}{\partial y}\frac{\partial N_j}{\partial y} \right) d\Omega
\end{equation}

\paragraph{}Como os gradientes das funções de forma ($\nabla N$) são constantes em elementos P1, a integral torna-se um produto direto pela área do elemento ($A_e$), resultando em uma montagem computacional extremamente eficiente \cite{Fish2007}.

\subsection{Matriz de Massa e Termo Transiente}
\paragraph{}Para equações dependentes do tempo (como o transporte de vorticidade), surge a matriz de massa $\mathbf{M}$, proveniente do termo $\frac{\partial \phi}{\partial t}$:

\begin{equation}
    M_{ij}^e = \int_{\Omega_e} N_i N_j \, d\Omega
\end{equation}

\paragraph{}Para triângulos lineares, esta integral resulta na matriz de massa consistente:
\begin{equation}
    \mathbf{M}^e = \frac{A_e}{12} \begin{bmatrix} 2 & 1 & 1 \\ 1 & 2 & 1 \\ 1 & 1 & 2 \end{bmatrix}
\end{equation}

\subsection{Tratamento do Termo Convectivo}
\paragraph{}O termo de advecção $\mathbf{u} \cdot \nabla \omega$ na Eq. (\ref{eq:transp_omega}) gera a matriz de convecção assimétrica $\mathbf{C}$:

\begin{equation}
    C_{ij}^e = \int_{\Omega_e} N_i (\mathbf{u} \cdot \nabla N_j) \, d\Omega
\end{equation}

\paragraph{}Observa-se que, para altos números de Reynolds, a dominância deste termo pode introduzir oscilações numéricas. Embora a implementação base utilize Galerkin padrão, a arquitetura permite a introdução de funções teste modificadas (Petrov-Galerkin) conforme proposto por Brooks e Hughes \cite{Brooks1982} para estabilização (SUPG).

\section{Discretização Temporal}

\paragraph{}Para integração no tempo, adota-se o esquema $\theta$-method (Diferenças Finitas no tempo):
\begin{equation}
    (\mathbf{M} + \theta \Delta t \mathbf{K}) \mathbf{\Phi}^{n+1} = (\mathbf{M} - (1-\theta) \Delta t \mathbf{K}) \mathbf{\Phi}^{n} + \Delta t \mathbf{F}
\end{equation}
\noindent O \textit{MinervaMesh} utiliza tipicamente $\theta=1$ (Euler Implícito) para garantir estabilidade incondicional em problemas difusivos, ou $\theta=0.5$ (Crank-Nicolson) para maior precisão de segunda ordem \cite{Versteeg2007}.
